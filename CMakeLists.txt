# UICC is the Windows SDK compiler for the native Windows Ribbon Framework.
# Generated BML/resources are build output; the application has no Python runtime.
get_filename_component(WINDOWS_KITS_ROOT
  "[HKEY_LOCAL_MACHINE\\SOFTWARE\\Microsoft\\Windows Kits\\Installed Roots;KitsRoot10]" ABSOLUTE)
file(GLOB UICC_SDK_HINTS LIST_DIRECTORIES true "${WINDOWS_KITS_ROOT}/bin/*/x86")
list(SORT UICC_SDK_HINTS COMPARE NATURAL ORDER DESCENDING)
find_program(UICC_EXECUTABLE NAMES uicc.exe HINTS
  "${WINDOWS_KITS_ROOT}/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x86"
  ${UICC_SDK_HINTS} REQUIRED)
set(RIBBON_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/ribbon")
file(MAKE_DIRECTORY "${RIBBON_GENERATED_DIR}")
add_custom_command(
  OUTPUT "${RIBBON_GENERATED_DIR}/ribbon.bml" "${RIBBON_GENERATED_DIR}/ribbon_ids.h" "${RIBBON_GENERATED_DIR}/ribbon.rc"
  # UICC appends to /res when the file already exists. Regeneration must start
  # with a fresh generated file, and one target must own the shared invocation.
  COMMAND "${CMAKE_COMMAND}" -E rm -f "${RIBBON_GENERATED_DIR}/ribbon.rc"
  COMMAND "${UICC_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/resources/ribbon.xml" ribbon.bml
    /header:ribbon_ids.h /res:ribbon.rc /name:APPLICATION
  DEPENDS resources/ribbon.xml
  WORKING_DIRECTORY "${RIBBON_GENERATED_DIR}"
  VERBATIM)
add_custom_target(explorer_ribbon_resources DEPENDS
  "${RIBBON_GENERATED_DIR}/ribbon.bml" "${RIBBON_GENERATED_DIR}/ribbon_ids.h" "${RIBBON_GENERATED_DIR}/ribbon.rc")
add_library(explorer_ribbon STATIC src/ribbon.cpp)
target_include_directories(explorer_ribbon PUBLIC include)
target_compile_definitions(explorer_ribbon PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(explorer_ribbon PUBLIC explorer_core ole32 shell32 shlwapi propsys uuid user32 advapi32 gdi32 oleacc uiautomationcore)
add_executable(WindowsExplorer WIN32 src/main.cpp src/app.cpp src/ribbon_app.cpp src/headless_smoke.cpp src/headless_benchmark.cpp src/headless_crash.cpp resources/app.rc)
target_sources(WindowsExplorer PRIVATE "${RIBBON_GENERATED_DIR}/ribbon.rc")
add_dependencies(WindowsExplorer explorer_ribbon_resources)
target_include_directories(WindowsExplorer PRIVATE "${RIBBON_GENERATED_DIR}")
target_link_libraries(WindowsExplorer PRIVATE explorer_core explorer_ribbon uxtheme mpr psapi)
if(MSVC)
  foreach(target explorer_core explorer_ribbon WindowsExplorer)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    # Retain optimized Release symbols for headless crash diagnosis. Symbols
    # are separate build output; no debugger or runtime dependency is shipped.
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:/Zi>")
  endforeach()
  target_link_options(WindowsExplorer PRIVATE /MANIFEST:NO)
  target_link_options(WindowsExplorer PRIVATE "$<$<CONFIG:Release>:/DEBUG:FULL>"
    "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>"
    "$<$<CONFIG:Release>:/INCREMENTAL:NO>" /PDBALTPATH:%_PDB%)
endif()
