cmake_minimum_required(VERSION 3.24)
project(WindowsExplorer VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows Explorer requires Windows 10 or newer and the Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(explorer_core STATIC src/core.cpp src/shell_operations.cpp src/search.cpp src/extra_operations.cpp src/input.cpp src/item_actions.cpp)
target_include_directories(explorer_core PUBLIC include)
target_compile_definitions(explorer_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(explorer_core PUBLIC ole32 oleaut32 shell32 shlwapi propsys uuid comctl32 user32 advapi32)
add_executable(WindowsExplorer WIN32 src/main.cpp src/app.cpp resources/app.rc)
target_link_libraries(WindowsExplorer PRIVATE explorer_core uxtheme mpr)
if(MSVC)
  foreach(target explorer_core WindowsExplorer)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endforeach()
  target_link_options(WindowsExplorer PRIVATE /MANIFEST:NO)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(explorer_tests tests/core_tests.cpp tests/shell_operations_tests.cpp tests/extra_operations_tests.cpp tests/input_tests.cpp tests/item_actions_tests.cpp tests/search_tests.cpp)
  target_link_libraries(explorer_tests PRIVATE explorer_core)
  if(MSVC)
    target_compile_options(explorer_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME core_and_shell_operations COMMAND explorer_tests)
  set_tests_properties(core_and_shell_operations PROPERTIES TIMEOUT 90)
  add_test(NAME hidden_shell_host COMMAND WindowsExplorer --headless-smoke --report ${CMAKE_BINARY_DIR}/headless-smoke.json)
  set_tests_properties(hidden_shell_host PROPERTIES TIMEOUT 60)
endif()
install(TARGETS WindowsExplorer RUNTIME DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
install(DIRECTORY docs DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME windows-explorer)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
